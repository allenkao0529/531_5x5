// 🚀 【妙招 2】前端樂觀更新 (Optimistic UI) ＋ 背景非同步盲發 function saveLog(rowIdx, movement) { const password = document.getElementById('passwordInput').value; if(!password) { alert('請先輸入密碼！'); return; } const weight = document.getElementById('modalActualWeight').value; const reps = document.getElementById('modalActualReps').value; if(!weight || !reps) { alert('請輸入實際重量與下數！'); return; } // 1. 【樂觀更新】直接在前端記憶體修改對應項目的狀態，確保日曆與畫面不會穿幫 if (currentWorkouts) { const item = currentWorkouts.find(w => w.rowIdx === rowIdx); if (item) { item.isDone = 'V'; item.actualWeight = weight; item.actualReps = reps; } } if (!allEvents[currentSelectedDate]) { allEvents[currentSelectedDate] = { move: movement, week: "", maxWeight: weight, sets: 1, done: true }; } else { allEvents[currentSelectedDate].done = true; if (Number(weight) > Number(allEvents[currentSelectedDate].maxWeight)) { allEvents[currentSelectedDate].maxWeight = weight; } } // 2. 瞬間重繪畫面，按鈕與打勾狀態 0 毫秒切換 renderWorkoutList(); renderCalendar(); // 3. 立即關閉輸入彈跳視窗，讓操作不間斷 const modalEl = document.getElementById('logModal'); const modal = bootstrap.Modal.getInstance(modalEl); if (modal) modal.hide(); // 4. 整理要傳送的 logData 資料結構（完全維持原設計） const logData = { rowIdx: rowIdx, date: currentSelectedDate, movement: movement, actualWeight: weight, actualReps: reps }; // 5. 【背景盲發】移除 await，丟給背景異步執行 console.log('前端更新完畢，連線正在背景傳送至試算表...'); callAPI({ action: 'logWorkout', password: password, logData: logData }) .then(res => { if (!res.success) { console.error('寫入失敗:', res.error); alert('記錄可能未成功同步至試算表，請重新整理網頁！原因: ' + res.error); } else { console.log('✅ 試算表寫入成功，快取已同步清空。'); } }) .catch(err => { console.error('網路錯誤:', err); }); }
